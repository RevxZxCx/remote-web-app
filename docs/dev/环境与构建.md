
### 手工探测接口（PowerShell 5.1 无 `-SkipHttpErrorCheck`，用 `HttpWebRequest`）

```powershell
function Probe($method,$url,$body){
  $r=[System.Net.HttpWebRequest]::Create($url); $r.Method=$method
  $r.ContentType='application/json'
  if($body){ $b=[Text.Encoding]::UTF8.GetBytes($body); $r.ContentLength=$b.Length
             $s=$r.GetRequestStream(); $s.Write($b,0,$b.Length); $s.Close() }
  try{ $resp=$r.GetResponse(); $sr=New-Object IO.StreamReader($resp.GetResponseStream())
       "$($resp.StatusCode) $($sr.ReadToEnd())" }
  catch [System.Net.WebException]{ $e=$_.Exception.Response
       if($e){ $sr=New-Object IO.StreamReader($e.GetResponseStream())
               "$([int]$e.StatusCode) $($sr.ReadToEnd())" } else { "ERR $($_.Exception.Message)" } }
}
Probe 'GET'  'http://127.0.0.1:19387/api/pair/status' $null
Probe 'POST' 'http://127.0.0.1:19387/api/session.list' '{}'
```
